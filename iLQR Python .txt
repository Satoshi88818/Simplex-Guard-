"""
stochastic_ilqr.py
--------------------
A from-scratch iLQR (iterative Linear Quadratic Regulator) trajectory
optimizer for a kinematic unicycle robot, augmented with a
STOCHASTIC RISK TERM that turns "epistemic uncertainty x occupancy"
into a real-time-tractable, differentiable addition to the running
cost -- this is Improvement #1 from the FSD v12 design document,
implemented and validated end to end.

Why this is the interesting piece to build
--------------------------------------------
A standard MPC/iLQR cost only knows about a *point estimate* of where
obstacles are (occupancy). It has no way to be more cautious in a
region the perception system is unsure about versus a region it is
confident is clear. The fix used here: maintain two scalar fields over
the workspace --

    occupancy(p)    in [0, 1]  : how likely something occupies point p
    uncertainty(p)  in [0, 1]  : how unsure the perception system is
                                  about that likelihood at point p

and add a risk cost term

    l_risk(p) = risk_weight * uncertainty(p) * occupancy(p)

to the running cost at every timestep along the trajectory. A region
with high occupancy but LOW uncertainty (a confidently-detected wall)
is treated as a normal hard-ish cost the planner will route around
anyway. A region with moderate occupancy AND high uncertainty (e.g.
"might be a pedestrian behind that parked van, sensor isn't sure") now
carries extra cost specifically because it's unsure -- pushing the
planner toward a route with a clearer, more confidently-assessed
margin, even when the point-estimate occupancy alone wouldn't have
forced a detour.

Both fields are represented as sums of smooth Gaussian "bumps," which
makes the risk term's gradient and Hessian available in closed form
(no autodiff framework, no finite differences) -- this directly
implements the design doc's stated refinement of replacing
finite-difference Jacobians with analytic ones, and the analytic
Jacobians for the *dynamics* are implemented the same way, for the
same reason: finite-differencing inside a real-time control loop is
expensive and noisy; closed-form derivatives are cheap and exact.

Everything below is validated against finite differences in the test
suite as an explicit correctness check, not just asserted.

Scope and honesty about limitations
--------------------------------------
- The dynamics model is a kinematic unicycle (state = x, y, heading;
  control = speed, turn-rate), not a full vehicle dynamics model with
  slip, tire forces, or actuator lag.
- The risk field is a synthetic sum of Gaussian bumps, not a real
  perception system's uncertainty/occupancy output. The point of this
  exercise is to validate that the MATH and the OPTIMIZER correctly
  turn "uncertainty x occupancy" into risk-averse routing -- not to
  claim this is a deployable perception interface.
- Obstacle avoidance here is soft (cost-based), not the hard
  constraint enforcement a real system would layer on top (that's
  what the Simplex Guard, built previously, is for).
"""

from __future__ import annotations

import math
from dataclasses import dataclass, field
from typing import Callable, List, Optional, Sequence, Tuple

import numpy as np

# ==========================================================================
# Scalar fields (uncertainty / occupancy), with analytic derivatives
# ==========================================================================


@dataclass(frozen=True)
class GaussianBump:
    """An isotropic Gaussian bump in 2D:

        f(p) = amplitude * exp(-||p - center||^2 / (2 * sigma^2))

    Closed-form gradient and Hessian (used directly by iLQR's cost
    quadratic expansion -- no autodiff, no finite differences):

        grad f(p) = f(p) * (-(p - center) / sigma^2)
        Hess f(p) = f(p) * ( outer(d, d) / sigma^4 - I / sigma^2 ),  d = p - center
    """

    center: Tuple[float, float]
    amplitude: float
    sigma: float

    def __post_init__(self):
        if self.sigma <= 0:
            raise ValueError("sigma must be > 0")

    def value(self, p: np.ndarray) -> float:
        d = p - np.asarray(self.center)
        return float(self.amplitude * math.exp(-float(d @ d) / (2.0 * self.sigma ** 2)))

    def gradient(self, p: np.ndarray) -> np.ndarray:
        d = p - np.asarray(self.center)
        f = self.value(p)
        return f * (-d / self.sigma ** 2)

    def hessian(self, p: np.ndarray) -> np.ndarray:
        d = p - np.asarray(self.center)
        f = self.value(p)
        outer = np.outer(d, d)
        return f * (outer / self.sigma ** 4 - np.eye(2) / self.sigma ** 2)


@dataclass
class ScalarField:
    """A sum of GaussianBumps, clipped to [0, 1] in VALUE ONLY for
    interpretation purposes (occupancy/uncertainty are probabilities).
    The clip is intentionally NOT applied inside gradient/Hessian --
    doing so would make the field non-differentiable at the clip
    boundary. In practice the bump amplitudes are chosen so the sum
    stays within [0, 1] almost everywhere for the scenarios used here;
    this is documented and tested explicitly (see
    `test_field_values_stay_in_unit_range_for_test_scenarios`).
    """

    bumps: List[GaussianBump] = field(default_factory=list)

    def value(self, p: np.ndarray) -> float:
        return float(sum(b.value(p) for b in self.bumps))

    def gradient(self, p: np.ndarray) -> np.ndarray:
        if not self.bumps:
            return np.zeros(2)
        return sum(b.gradient(p) for b in self.bumps)

    def hessian(self, p: np.ndarray) -> np.ndarray:
        if not self.bumps:
            return np.zeros((2, 2))
        return sum(b.hessian(p) for b in self.bumps)


@dataclass
class RiskField:
    """risk(p) = uncertainty(p) * occupancy(p), with exact gradient and
    Hessian via the product rule:

        h = u * o
        grad h  = u * grad_o + o * grad_u
        Hess h  = u * Hess_o + o * Hess_u
                  + outer(grad_u, grad_o) + outer(grad_o, grad_u)
    """

    uncertainty: ScalarField
    occupancy: ScalarField

    def value(self, p: np.ndarray) -> float:
        return self.uncertainty.value(p) * self.occupancy.value(p)

    def gradient(self, p: np.ndarray) -> np.ndarray:
        u = self.uncertainty.value(p)
        o = self.occupancy.value(p)
        gu = self.uncertainty.gradient(p)
        go = self.occupancy.gradient(p)
        return u * go + o * gu

    def hessian(self, p: np.ndarray) -> np.ndarray:
        u = self.uncertainty.value(p)
        o = self.occupancy.value(p)
        gu = self.uncertainty.gradient(p)
        go = self.occupancy.gradient(p)
        hu = self.uncertainty.hessian(p)
        ho = self.occupancy.hessian(p)
        return u * ho + o * hu + np.outer(gu, go) + np.outer(go, gu)


# ==========================================================================
# Dynamics: kinematic unicycle, with analytic Jacobians
# ==========================================================================

STATE_DIM = 3  # [x, y, heading]
CONTROL_DIM = 2  # [speed, turn_rate]


def unicycle_step(x: np.ndarray, u: np.ndarray, dt: float) -> np.ndarray:
    """x_{t+1} = f(x_t, u_t):
        x'     = x + v * cos(heading) * dt
        y'     = y + v * sin(heading) * dt
        heading' = heading + omega * dt
    """
    px, py, heading = x
    v, omega = u
    return np.array(
        [
            px + v * math.cos(heading) * dt,
            py + v * math.sin(heading) * dt,
            heading + omega * dt,
        ]
    )


def unicycle_jacobians(x: np.ndarray, u: np.ndarray, dt: float) -> Tuple[np.ndarray, np.ndarray]:
    """Analytic df/dx (A, 3x3) and df/du (B, 3x2) for the unicycle step.

        df/dx = [[1, 0, -v*sin(h)*dt],
                 [0, 1,  v*cos(h)*dt],
                 [0, 0,  1          ]]

        df/du = [[cos(h)*dt, 0],
                 [sin(h)*dt, 0],
                 [0,         dt]]
    """
    _, _, heading = x
    v, _ = u
    A = np.array(
        [
            [1.0, 0.0, -v * math.sin(heading) * dt],
            [0.0, 1.0, v * math.cos(heading) * dt],
            [0.0, 0.0, 1.0],
        ]
    )
    B = np.array(
        [
            [math.cos(heading) * dt, 0.0],
            [math.sin(heading) * dt, 0.0],
            [0.0, dt],
        ]
    )
    return A, B


# ==========================================================================
# Cost function
# ==========================================================================


@dataclass
class CostConfig:
    Q_running: np.ndarray  # 3x3, running state cost (usually small/zero)
    R: np.ndarray  # 2x2, control effort cost
    Qf: np.ndarray  # 3x3, terminal state cost (drives goal-reaching)
    goal: np.ndarray  # (3,) goal state
    risk_weight: float = 0.0
    risk_field: Optional[RiskField] = None

    def stage_cost(self, x: np.ndarray, u: np.ndarray) -> float:
        dx = x - self.goal
        cost = 0.5 * dx @ self.Q_running @ dx + 0.5 * u @ self.R @ u
        if self.risk_weight > 0 and self.risk_field is not None:
            cost += self.risk_weight * self.risk_field.value(x[:2])
        return float(cost)

    def terminal_cost(self, x: np.ndarray) -> float:
        dx = x - self.goal
        return float(0.5 * dx @ self.Qf @ dx)

    def stage_derivatives(
        self, x: np.ndarray, u: np.ndarray
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Returns (l_x, l_u, l_xx, l_uu, l_ux)."""
        dx = x - self.goal
        l_x = self.Q_running @ dx
        l_u = self.R @ u
        l_xx = self.Q_running.copy()
        l_uu = self.R.copy()
        l_ux = np.zeros((CONTROL_DIM, STATE_DIM))

        if self.risk_weight > 0 and self.risk_field is not None:
            p = x[:2]
            g = self.risk_field.gradient(p)  # (2,)
            H = self.risk_field.hessian(p)  # (2,2)
            l_x = l_x.copy()
            l_x[:2] += self.risk_weight * g
            l_xx = l_xx.copy()
            l_xx[:2, :2] += self.risk_weight * H
            # No u- or heading-dependence in the risk term, so l_u,
            # l_uu, l_ux get no contribution from it.

        return l_x, l_u, l_xx, l_uu, l_ux

    def terminal_derivatives(self, x: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        dx = x - self.goal
        return self.Qf @ dx, self.Qf.copy()


# ==========================================================================
# iLQR solver
# ==========================================================================


@dataclass
class ILQRResult:
    X: np.ndarray  # (T+1, 3)
    U: np.ndarray  # (T, 2)
    cost_history: List[float]
    converged: bool
    iterations: int
    final_regularization: float


class ILQRSolver:
    def __init__(
        self,
        cost: CostConfig,
        dt: float = 0.1,
        max_iters: int = 100,
        cost_tol: float = 1e-5,
        reg_init: float = 1.0,
        reg_min: float = 1e-6,
        reg_max: float = 1e10,
        reg_factor: float = 2.0,
        line_search_alphas: Optional[Sequence[float]] = None,
        max_control: Optional[np.ndarray] = None,
    ):
        self.cost = cost
        self.dt = dt
        self.max_iters = max_iters
        self.cost_tol = cost_tol
        self.reg_init = reg_init
        self.reg_min = reg_min
        self.reg_max = reg_max
        self.reg_factor = reg_factor
        self.line_search_alphas = list(line_search_alphas or [1.0, 0.5, 0.25, 0.1, 0.05, 0.01])
        self.max_control = max_control

    # -- helpers ---------------------------------------------------------

    def _rollout(self, x0: np.ndarray, U: np.ndarray) -> np.ndarray:
        T = U.shape[0]
        X = np.zeros((T + 1, STATE_DIM))
        X[0] = x0
        for t in range(T):
            X[t + 1] = unicycle_step(X[t], U[t], self.dt)
        return X

    def _total_cost(self, X: np.ndarray, U: np.ndarray) -> float:
        T = U.shape[0]
        total = sum(self.cost.stage_cost(X[t], U[t]) for t in range(T))
        total += self.cost.terminal_cost(X[T])
        return total

    def _clip_control(self, u: np.ndarray) -> np.ndarray:
        if self.max_control is None:
            return u
        return np.clip(u, -self.max_control, self.max_control)

    # -- backward pass -----------------------------------------------------

    def _backward_pass(
        self, X: np.ndarray, U: np.ndarray, reg: float
    ) -> Optional[Tuple[List[np.ndarray], List[np.ndarray], float]]:
        """Returns (k_list, K_list, expected_improvement) or None if
        Q_uu could not be made positive-definite even at the current
        regularization (caller should increase reg and retry).

        `expected_improvement` is the standard DDP/iLQR predicted cost
        reduction  sum_t [ -(Q_u^T k_t + 0.5 * k_t^T Q_uu k_t) ]  --
        computed directly from the backward pass, independent of
        whether a forward-pass line search actually finds an
        improving step. This is what lets the solver correctly
        recognize convergence when the cost is ALREADY at (or
        essentially at) its minimum: in that case Q_u ~= 0 everywhere,
        expected_improvement ~= 0, and no line-search step could ever
        reduce a cost that is already minimal -- that is a sign of
        convergence, not a sign that regularization needs to keep
        increasing forever.
        """
        T = U.shape[0]
        Vx, Vxx = self.cost.terminal_derivatives(X[T])

        k_list: List[np.ndarray] = [None] * T  # type: ignore
        K_list: List[np.ndarray] = [None] * T  # type: ignore
        expected_improvement = 0.0

        for t in reversed(range(T)):
            x, u = X[t], U[t]
            A, B = unicycle_jacobians(x, u, self.dt)
            l_x, l_u, l_xx, l_uu, l_ux = self.cost.stage_derivatives(x, u)

            Q_x = l_x + A.T @ Vx
            Q_u = l_u + B.T @ Vx
            Q_xx = l_xx + A.T @ Vxx @ A
            Q_uu = l_uu + B.T @ Vxx @ B
            Q_ux = l_ux + B.T @ Vxx @ A

            Q_uu_reg = Q_uu + reg * np.eye(CONTROL_DIM)

            # Positive-definiteness check via Cholesky; signals the
            # caller to increase regularization and retry the WHOLE
            # backward pass if it fails partway through (standard
            # DDP/iLQR regularization scheme).
            try:
                L = np.linalg.cholesky(Q_uu_reg)
            except np.linalg.LinAlgError:
                return None

            Q_uu_inv = np.linalg.inv(Q_uu_reg)
            k = -Q_uu_inv @ Q_u
            K = -Q_uu_inv @ Q_ux

            expected_improvement += -(Q_u @ k + 0.5 * k @ Q_uu @ k)

            Vx = Q_x + K.T @ Q_uu @ k + K.T @ Q_u + Q_ux.T @ k
            Vxx = Q_xx + K.T @ Q_uu @ K + K.T @ Q_ux + Q_ux.T @ K
            Vxx = 0.5 * (Vxx + Vxx.T)  # enforce symmetry against fp drift

            k_list[t] = k
            K_list[t] = K

        return k_list, K_list, expected_improvement

    # -- forward pass with line search -------------------------------------

    def _forward_pass(
        self,
        X: np.ndarray,
        U: np.ndarray,
        k_list: List[np.ndarray],
        K_list: List[np.ndarray],
        current_cost: float,
    ) -> Optional[Tuple[np.ndarray, np.ndarray, float]]:
        T = U.shape[0]
        for alpha in self.line_search_alphas:
            X_new = np.zeros_like(X)
            U_new = np.zeros_like(U)
            X_new[0] = X[0]
            for t in range(T):
                dx = X_new[t] - X[t]
                u = U[t] + alpha * k_list[t] + K_list[t] @ dx
                u = self._clip_control(u)
                U_new[t] = u
                X_new[t + 1] = unicycle_step(X_new[t], u, self.dt)
            new_cost = self._total_cost(X_new, U_new)
            if np.isfinite(new_cost) and new_cost < current_cost:
                return X_new, U_new, new_cost
        return None

    # -- main solve loop -----------------------------------------------------

    def solve(self, x0: np.ndarray, U_init: np.ndarray) -> ILQRResult:
        X = self._rollout(x0, U_init)
        U = U_init.copy()
        cost = self._total_cost(X, U)
        cost_history = [cost]
        reg = self.reg_init
        converged = False
        it = 0

        for it in range(1, self.max_iters + 1):
            backward = None
            # Increase regularization until the backward pass succeeds
            # (Q_uu positive-definite at every timestep) or we give up.
            while backward is None:
                backward = self._backward_pass(X, U, reg)
                if backward is None:
                    reg *= self.reg_factor
                    if reg > self.reg_max:
                        return ILQRResult(X, U, cost_history, False, it, reg)
            k_list, K_list, expected_improvement = backward

            # Convergence check BEFORE attempting a line search: if the
            # backward pass itself predicts negligible further
            # improvement is possible (e.g. we are already at, or
            # extremely close to, a local optimum -- Q_u ~= 0
            # everywhere), no forward-pass step could ever succeed by
            # construction (there is nothing left to improve), and
            # that is a sign of CONVERGENCE, not a sign that
            # regularization needs to keep increasing. Without this
            # check, a problem that starts exactly at its optimum
            # (e.g. goal == start) would spin regularization up to
            # reg_max and incorrectly report failure.
            if expected_improvement < self.cost_tol:
                converged = True
                break

            result = self._forward_pass(X, U, k_list, K_list, cost)
            if result is None:
                # No line-search step improved cost: increase
                # regularization (be more conservative) and retry.
                reg *= self.reg_factor
                if reg > self.reg_max:
                    break
                continue

            X_new, U_new, new_cost = result
            improvement = cost - new_cost
            X, U, cost = X_new, U_new, new_cost
            cost_history.append(cost)
            reg = max(reg / self.reg_factor, self.reg_min)

            if improvement < self.cost_tol:
                converged = True
                break

        return ILQRResult(X, U, cost_history, converged, it, reg)
