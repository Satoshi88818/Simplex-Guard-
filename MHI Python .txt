"""
multi_hypothesis_ilqr.py
---------------------------
Fixes the symmetric-saddle limitation documented and measured in the
previous two deliverables: a perfectly symmetric obstacle placed
exactly on the straight-line path creates a true local-optimization
saddle, and closed-loop replanning alone does NOT resolve it (verified
directly, not assumed, in the full-stack-integration deliverable) --
every cycle re-solves the same symmetric problem the same way and
gets the same non-avoiding answer.

The fix here is the smallest thing that actually closes the gap: solve
the SAME iLQR problem multiple times per cycle from a small set of
different symmetry-breaking initial guesses (e.g. "assume I should
lean left", "assume I should lean right", plus the ordinary
unperturbed guess), and keep whichever converges to the lowest cost.
This is a standard, cheap way to sidestep a local optimizer's
sensitivity to its starting point -- it does not fix the underlying
non-convexity, it just makes the failure mode require a considerably
more adversarial scenario to actually trigger (quantified honestly in
Section 5 of the accompanying write-up, not claimed to be solved for
every possible case).

Nothing in simplex_guard.py, closed_loop_sim.py, stochastic_ilqr.py,
or full_stack_integration.py is modified. This file only adds a new
planner that is a drop-in alternative to ILQRPlanner.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import List, Optional, Sequence, Tuple

import numpy as np

from full_stack_integration import CURVATURE_SPEED_EPS, ILQRPlanner, ILQRPlannerConfig
from closed_loop_sim import VehicleState
from stochastic_ilqr import CostConfig, ILQRSolver


def _solve_cycle(
    cfg: ILQRPlannerConfig,
    risk_helper: ILQRPlanner,
    ego: VehicleState,
    t: float,
    T: int,
    dt: float,
    visible_actors_world: Sequence[Tuple[float, float, float, float]],
    U_init: np.ndarray,
) -> Tuple[np.ndarray, np.ndarray, float]:
    """Runs one iLQR solve from a given initial control guess and
    returns (U, X, final_cost). Shared by ILQRPlanner and
    MultiHypothesisILQRPlanner so both build the exact same cost
    problem from the exact same risk-field construction logic --
    `risk_helper` is a throwaway ILQRPlanner instance used ONLY for
    its (pure, state-free) `_build_risk_field` method, never for its
    own warm-start state.
    """
    risk_field = risk_helper._build_risk_field(ego, visible_actors_world)
    goal_local = np.array([cfg.target_speed_mps * T * dt, 0.0, 0.0])
    cost = CostConfig(
        Q_running=np.zeros((3, 3)),
        R=np.diag(cfg.control_cost),
        Qf=np.diag(
            [cfg.terminal_position_weight, cfg.terminal_position_weight, cfg.terminal_heading_weight]
        ),
        goal=goal_local,
        risk_weight=cfg.risk_weight if risk_field is not None else 0.0,
        risk_field=risk_field,
    )
    solver = ILQRSolver(
        cost,
        dt=dt,
        max_iters=cfg.max_ilqr_iters,
        cost_tol=cfg.cost_tol,
        max_control=np.array([cfg.max_speed_mps, cfg.max_turn_rate]),
    )
    x0_local = np.zeros(3)
    result = solver.solve(x0_local, U_init)
    final_cost = result.cost_history[-1]
    return result.U, result.X, final_cost


def _to_state_trajectory(U: np.ndarray, X: np.ndarray, ego: VehicleState, T: int) -> np.ndarray:
    """Same state-trajectory conversion used by ILQRPlanner.generate
    (see full_stack_integration.py for the two-bugs-found write-up on
    why this specific index-0 handling matters): index 0's SPEED is
    the ego's actual current speed (physical continuity), index 0's
    CURVATURE is the solver's fresh first-control decision (directly
    commandable, no continuity constraint).
    """
    traj = np.zeros((T, 5))
    v0, omega0 = U[0]
    curvature0 = omega0 / v0 if abs(v0) > CURVATURE_SPEED_EPS else 0.0
    traj[0] = [0.0, 0.0, 0.0, ego.speed, curvature0]
    for t_idx in range(1, T):
        v, omega = U[t_idx - 1]
        curvature = omega / v if abs(v) > CURVATURE_SPEED_EPS else 0.0
        traj[t_idx] = [X[t_idx, 0], X[t_idx, 1], X[t_idx, 2], v, curvature]
    return traj


@dataclass
class MultiHypothesisConfig:
    inner: ILQRPlannerConfig = field(default_factory=ILQRPlannerConfig)
    # Turn-rate nudges applied to a FRESH (non-warm-started) initial
    # guess to give each hypothesis a distinct starting basin. 0.0 is
    # always included implicitly unless explicitly excluded -- the
    # ordinary unperturbed guess is itself a legitimate hypothesis and
    # is usually the winner in the (common) non-symmetric case.
    nudges: Tuple[float, ...] = (-0.15, 0.0, 0.15)
    nudge_step_index: int = 5  # which horizon step gets the nudge (matches the convention used to break symmetry in the stochastic-iLQR deliverable's own test scenarios)


class MultiHypothesisILQRPlanner:
    """Drop-in alternative to ILQRPlanner: solves `len(nudges)`
    independent iLQR problems per cycle (one per symmetry-breaking
    initial guess), each warm-started from ITS OWN previous cycle's
    solution (not the previous winner's -- see write-up for why), and
    executes whichever converges to the lowest total cost this cycle.

    Diagnostics exposed after each `generate()` call:
        last_hypothesis_costs : List[float]  -- one per hypothesis, in
            the same order as `config.nudges`
        last_winner_index     : int          -- which one was executed
        last_solve_count      : int          -- == len(config.nudges),
            exposed directly so the per-cycle compute cost is easy to
            assert on rather than inferred
    """

    def __init__(self, config: Optional[MultiHypothesisConfig] = None):
        self.config = config or MultiHypothesisConfig()
        self._risk_helper = ILQRPlanner(self.config.inner)  # only its _build_risk_field is used
        n = len(self.config.nudges)
        self._U_prev_per_hypothesis: List[Optional[np.ndarray]] = [None] * n
        self.last_hypothesis_costs: Optional[List[float]] = None
        self.last_winner_index: Optional[int] = None
        self.last_solve_count: int = 0

    def reset(self) -> None:
        n = len(self.config.nudges)
        self._U_prev_per_hypothesis = [None] * n
        self.last_hypothesis_costs = None
        self.last_winner_index = None

    def _initial_guess(self, hypothesis_idx: int, T: int, dt: float) -> np.ndarray:
        cfg = self.config.inner
        U_prev = self._U_prev_per_hypothesis[hypothesis_idx]
        if U_prev is not None and U_prev.shape[0] == T:
            # Warm start from THIS hypothesis's own previous solution,
            # shifted forward one step -- each hypothesis tracks its
            # own local optimum over time, independent of whichever
            # one happened to win last cycle.
            return np.vstack([U_prev[1:], U_prev[-1:]])
        U_init = np.tile(np.array([cfg.target_speed_mps, 0.0]), (T, 1))
        nudge = self.config.nudges[hypothesis_idx]
        idx = min(self.config.nudge_step_index, T - 1) if T > 0 else 0
        if T > 0 and nudge != 0.0:
            U_init[idx, 1] += nudge
        return U_init

    def generate(
        self,
        ego: VehicleState,
        last_curvature: float,
        t: float,
        T: int,
        dt: float,
        visible_actors_world: Sequence[Tuple[float, float, float, float]],
    ) -> np.ndarray:
        results = []
        for idx in range(len(self.config.nudges)):
            U_init = self._initial_guess(idx, T, dt)
            U_res, X_res, cost_res = _solve_cycle(
                self.config.inner, self._risk_helper, ego, t, T, dt, visible_actors_world, U_init
            )
            self._U_prev_per_hypothesis[idx] = U_res.copy()
            results.append((U_res, X_res, cost_res))

        costs = [r[2] for r in results]
        winner_idx = int(np.argmin(costs))
        self.last_hypothesis_costs = costs
        self.last_winner_index = winner_idx
        self.last_solve_count = len(results)

        U_win, X_win, _ = results[winner_idx]
        return _to_state_trajectory(U_win, X_win, ego, T)
