"""
bicycle_guard_integration.py
--------------------------------
Reconnects the Simplex Guard lineage (simplex_guard.py,
stateful_simplex_guard.py, debounced_simplex_guard.py -- all built and
tested exclusively against the kinematic unicycle) with the real
vehicle dynamics lineage (bicycle_dynamics_tire_model.py,
combined_slip_tire_model.py, neural_pacejka.py). These two lineages
have run in parallel since the original full-stack-integration
deliverable and have never been reunited.

THE CORE PROBLEM THIS SOLVES: every guard's fallback response
(`simplex_guard.generate_stop_trajectory`) assumes a simple
`(x, y, heading, speed, curvature)` kinematic state with directly-
commandable curvature -- exactly the unicycle's world. The bicycle
model's state is 6-dimensional with REAL INERTIA (lateral velocity
`vy` and yaw rate `r` cannot reset instantly, unlike a kinematic
curvature label), and its control is `[delta, Fxf, Fxr]`, not
`[speed, curvature]`. Naively reusing the guard's own kinematic
fallback trajectory as if it were directly executable by the bicycle
model would silently assume physically impossible things -- e.g. that
a vehicle mid-corner with real lateral velocity can simply stop
turning the instant the guard decides to.

THE FIX: `BicycleGuardAdapter` translates in both directions --
bicycle state/trajectory -> the guard's kinematic format (so the
guard's own TTC/stopping-distance/lateral-accel checks, which only
ever needed `speed` and `curvature`, still work unchanged) -- and,
when the guard decides to override, translates the guard's DECISION
(which fallback tier: emergency stop, comfort stop, or cornering-speed
limit) back into a REAL, dynamically-feasible bicycle control
sequence, generated by actually rolling the combined-slip dynamics
forward, not by assuming idealized kinematics.

Nothing in simplex_guard.py, stateful_simplex_guard.py,
debounced_simplex_guard.py, bicycle_dynamics_tire_model.py,
combined_slip_tire_model.py, or neural_pacejka.py is modified. The
guard passed to `BicycleGuardAdapter` can be a plain SimplexGuard, a
StatefulSimplexGuard, or a DebouncedSimplexGuard -- all three expose
the identical `check_and_override` signature this adapter calls, by
design, all the way back to the original guard deliverable.
"""

from __future__ import annotations

import math
from dataclasses import dataclass
from typing import Optional, Tuple

import numpy as np

from simplex_guard import FallbackMode, GuardDecision, ViolationType, max_safe_cornering_speed
from bicycle_dynamics_tire_model import VehicleParams
from combined_slip_tire_model import (
    CONTROL_DIM,
    STATE_DIM,
    combined_slip_bicycle_step,
)
from neural_pacejka import NeuralPacejkaTireModel, PacejkaParams

CURVATURE_VX_FLOOR = 0.5  # below this forward speed, curvature is treated as 0 (matches slip_angles' own vx_floor convention)


def _sanitize_bicycle_state(state: np.ndarray) -> np.ndarray:
    """Replaces any non-finite entry of a bicycle state with a
    conservative default -- used ONLY when the guard has already
    flagged the ego state as untrustworthy (INVALID_SENSOR_INPUT), so
    the real-dynamics fallback generator has something finite to roll
    forward from instead of propagating NaN through the whole
    trajectory. Position/heading default to 0 if invalid (no reliable
    information to do otherwise); forward speed defaults to 0 (the
    safest assumption -- never assume unverified high speed); lateral
    velocity/yaw rate default to 0 for the same reason. This mirrors
    the base guard's own sanitization approach in its INVALID_SENSOR_INPUT
    handling (simplex_guard.py), applied here to the additional state
    dimensions the bicycle model has that the guard's own kinematic
    state never needed to sanitize.
    """
    x, y, heading, vx, vy, r = state
    x = x if math.isfinite(x) else 0.0
    y = y if math.isfinite(y) else 0.0
    heading = heading if math.isfinite(heading) else 0.0
    vx = vx if math.isfinite(vx) and vx >= 0.0 else 0.0
    vy = vy if math.isfinite(vy) else 0.0
    r = r if math.isfinite(r) else 0.0
    return np.array([x, y, heading, vx, vy, r])


# ==========================================================================
# State/trajectory format conversion: bicycle (6-dim) <-> guard (5-dim)
# ==========================================================================


def bicycle_state_to_guard_state(state: np.ndarray) -> np.ndarray:
    """[x, y, heading, vx, vy, r] -> [x, y, heading, speed, curvature].

    speed = hypot(vx, vy): the vehicle's TOTAL velocity magnitude, not
    just its forward component -- a car sliding sideways is still
    moving at that speed even if vx alone understates it, and the
    guard's TTC/stopping-distance checks are about closing speed on an
    obstacle, which cares about total motion, not just the
    forward-axis component.

    curvature = r / vx (yaw rate / forward speed, the same convention
    used by every previous adapter in this series -- e.g. the
    ILQRPlanner in full_stack_integration.py computed curvature =
    omega/v for the unicycle the same way), floored at
    CURVATURE_VX_FLOOR to avoid a division blowup at near-zero forward
    speed (curvature is physically undefined there, and also
    irrelevant to the guard's lateral-accel check, which multiplies by
    speed^2 -- a near-zero speed makes that product vanish regardless
    of curvature's value, so flooring to 0 here costs nothing safety-
    relevant).
    """
    x, y, heading, vx, vy, r = state
    speed = math.hypot(vx, vy)
    vx_eff = vx if abs(vx) > CURVATURE_VX_FLOOR else (CURVATURE_VX_FLOOR if vx >= 0 else -CURVATURE_VX_FLOOR)
    curvature = r / vx_eff if abs(vx) > CURVATURE_VX_FLOOR else 0.0
    return np.array([x, y, heading, speed, curvature])


def bicycle_trajectory_to_guard_trajectory(state_traj: np.ndarray) -> np.ndarray:
    """(T, 6) bicycle state trajectory -> (T, 5) guard-format
    trajectory, converting each state independently. Callers are
    responsible for the same index-0-is-the-actual-current-state
    convention established by every previous trajectory-format adapter
    in this series (the guard's own fallback generators, and the
    unicycle ILQRPlanner in full_stack_integration.py, both rely on
    this convention).
    """
    T = state_traj.shape[0]
    guard_traj = np.zeros((T, 5))
    for t in range(T):
        guard_traj[t] = bicycle_state_to_guard_state(state_traj[t])
    return guard_traj


# ==========================================================================
# Dynamically-feasible fallback generators (real dynamics, not kinematics)
# ==========================================================================


def generate_bicycle_stop_trajectory(
    ego_state: np.ndarray,
    decel_target_mps2: float,
    horizon_steps: int,
    dt: float,
    tire_model: NeuralPacejkaTireModel,
    pacejka_params: PacejkaParams,
    vp: VehicleParams,
    front_brake_bias: float = 0.6,
) -> Tuple[np.ndarray, np.ndarray]:
    """Generates a straight-line (delta=0) braking maneuver by
    actually rolling the combined-slip dynamics forward with a
    constant commanded deceleration, split between axles by
    `front_brake_bias` (fraction of total braking force at the front
    -- 0.6 is a representative front-biased split, typical of real
    brake proportioning). Returns (state_traj (horizon_steps, 6),
    control_traj (horizon_steps, 3)).

    Unlike the guard's own kinematic `generate_stop_trajectory` (which
    assumes speed decreases at exactly `decel_target_mps2` with no
    other coupling), this ACTUALLY INTEGRATES the real dynamics --
    including whatever residual lateral velocity/yaw rate the vehicle
    had when the override began, which does NOT reset instantly just
    because delta is now commanded to 0. See the headline scenario in
    the accompanying write-up for a concrete demonstration of why this
    distinction matters.
    """
    target_Fx_total = -abs(decel_target_mps2) * vp.mass_kg  # braking = negative Fx
    Fxf = target_Fx_total * front_brake_bias
    Fxr = target_Fx_total * (1.0 - front_brake_bias)
    control = np.array([0.0, Fxf, Fxr])

    state = np.array(ego_state, dtype=np.float64)
    state_traj = np.zeros((horizon_steps, STATE_DIM))
    control_traj = np.zeros((horizon_steps, CONTROL_DIM))
    for t in range(horizon_steps):
        state_traj[t] = state
        control_traj[t] = control
        state = combined_slip_bicycle_step(state, control, tire_model, pacejka_params, vp, dt)
        state[3] = max(state[3], 0.0)  # forward speed cannot go negative under braking
    return state_traj, control_traj


def generate_bicycle_speed_limited_trajectory(
    candidate_state_traj: np.ndarray,
    candidate_control_traj: np.ndarray,
    safe_speed_mps: float,
    tire_model: NeuralPacejkaTireModel,
    pacejka_params: PacejkaParams,
    vp: VehicleParams,
    dt: float,
    decel_mps2: float = 3.0,
) -> Tuple[np.ndarray, np.ndarray]:
    """Keeps the CANDIDATE plan's own steering (delta) sequence --
    the path itself isn't the problem, the speed on that path is --
    and replaces only the longitudinal force with a comfort-level
    braking command whenever current speed exceeds `safe_speed_mps`,
    rolled forward through the real dynamics so the resulting
    lateral behavior on that (now slower) path is physically honest,
    not just relabeled.
    """
    T = candidate_control_traj.shape[0]
    state = np.array(candidate_state_traj[0], dtype=np.float64)
    state_traj = np.zeros((T, STATE_DIM))
    control_traj = np.zeros((T, CONTROL_DIM))
    for t in range(T):
        state_traj[t] = state
        delta = candidate_control_traj[t, 0]
        if state[3] > safe_speed_mps:
            Fx_total = -abs(decel_mps2) * vp.mass_kg
        else:
            Fx_total = 0.0
        control = np.array([delta, Fx_total * 0.5, Fx_total * 0.5])
        control_traj[t] = control
        state = combined_slip_bicycle_step(state, control, tire_model, pacejka_params, vp, dt)
        state[3] = max(state[3], 0.0)
    return state_traj, control_traj


# ==========================================================================
# The adapter itself
# ==========================================================================


@dataclass
class BicycleGuardConfig:
    emergency_decel_mps2: float = 7.0
    comfort_decel_mps2: float = 3.0
    front_brake_bias: float = 0.6


class BicycleGuardAdapter:
    """Wraps any of SimplexGuard / StatefulSimplexGuard /
    DebouncedSimplexGuard (duck-typed via their shared
    `check_and_override` signature) and makes it usable with the
    bicycle+combined-slip dynamics instead of the kinematic unicycle.
    """

    def __init__(
        self,
        guard,
        tire_model: NeuralPacejkaTireModel,
        pacejka_params: PacejkaParams,
        vp: VehicleParams,
        config: Optional[BicycleGuardConfig] = None,
    ):
        self.guard = guard
        self.tire_model = tire_model
        self.pacejka_params = pacejka_params
        self.vp = vp
        self.config = config or BicycleGuardConfig()

    def check_and_override(
        self,
        candidate_state_traj: np.ndarray,
        candidate_control_traj: np.ndarray,
        ego_state: np.ndarray,
        occupancy: Optional[np.ndarray],
        dt: float,
        resolution_m: Optional[float] = None,
        mu: Optional[float] = None,
        occupancy_origin: Optional[Tuple[int, int]] = None,
    ) -> Tuple[GuardDecision, np.ndarray, np.ndarray, bool]:
        guard_ego_state = bicycle_state_to_guard_state(ego_state)
        guard_candidate_traj = bicycle_trajectory_to_guard_trajectory(candidate_state_traj)

        decision, _, overridden = self.guard.check_and_override(
            guard_candidate_traj, guard_ego_state, occupancy, dt, resolution_m, mu, occupancy_origin
        )

        if not overridden:
            return decision, candidate_state_traj, candidate_control_traj, False

        cfg = self.config
        if decision.fallback_mode == FallbackMode.SPEED_LIMIT_CORNER:
            worst_curvature = float(np.max(np.abs(guard_candidate_traj[:, 4])))
            mu_val = mu if mu is not None else self.guard.config.friction_default
            max_lat_accel = self.guard.config.max_lateral_accel_mps2(mu_val)
            safe_speed = max_safe_cornering_speed(worst_curvature, max_lat_accel)
            safe_state_traj, safe_control_traj = generate_bicycle_speed_limited_trajectory(
                candidate_state_traj, candidate_control_traj, safe_speed,
                self.tire_model, self.pacejka_params, self.vp, dt,
                decel_mps2=cfg.comfort_decel_mps2,
            )
        else:
            # EMERGENCY_STOP, COMFORT_STOP, or (fail-safe default for
            # any other overridden case, e.g. INVALID_SENSOR_INPUT,
            # which the base guard already always maps to
            # EMERGENCY_STOP -- this branch just mirrors that).
            decel = (
                cfg.emergency_decel_mps2
                if decision.fallback_mode == FallbackMode.EMERGENCY_STOP
                else cfg.comfort_decel_mps2
            )
            T = candidate_control_traj.shape[0]
            # If the ego state itself was flagged untrustworthy, don't
            # roll real dynamics forward from a NaN-contaminated
            # starting point -- sanitize it first (see
            # _sanitize_bicycle_state's docstring for the reasoning).
            fallback_start_state = (
                _sanitize_bicycle_state(ego_state)
                if ViolationType.INVALID_SENSOR_INPUT in decision.violations
                else ego_state
            )
            safe_state_traj, safe_control_traj = generate_bicycle_stop_trajectory(
                fallback_start_state, decel, T, dt, self.tire_model, self.pacejka_params, self.vp,
                front_brake_bias=cfg.front_brake_bias,
            )

        return decision, safe_state_traj, safe_control_traj, True
